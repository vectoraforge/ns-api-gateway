DSN = "postgres://%s:%s@%s:%s/%s" % (os.environ["DB_USER"], os.environ["DB_PASSWORD"], os.environ["DB_HOST"], os.environ["DB_PORT"], os.environ["DB_NAME"])

asyncio.run(main())
