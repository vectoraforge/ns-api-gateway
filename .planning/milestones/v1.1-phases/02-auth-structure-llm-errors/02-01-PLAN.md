```python
# app/auth.py — current state (Phase 1 output)
import base64
import json
import time

from fastapi import Header

from app.exceptions import MissingTokenError, InvalidTokenError, ExpiredTokenError


def _decode_jwt_payload(token: str) -> dict:
    parts = token.split(".")
    if len(parts) < 2:
        raise ValueError("Invalid token")
    payload_b64 = parts[1]
    payload_b64 += "=" * (-len(payload_b64) % 4)
    raw = base64.urlsafe_b64decode(payload_b64.encode("utf-8"))
    return json.loads(raw.decode("utf-8"))


async def get_user_id(authorization: str | None = Header(None)) -> str:
    if not authorization:
        raise MissingTokenError()
    if not authorization.startswith("Bearer "):
        raise MissingTokenError()
    token = authorization.split(" ", 1)[1].strip()
    if not token:
        raise MissingTokenError()
    try:
        payload = _decode_jwt_payload(token)
    except Exception:
        raise InvalidTokenError() from None
    exp = payload.get("exp")
    if exp is not None and exp < time.time():
        raise ExpiredTokenError()
    user_id = payload.get("user_id")
    if not user_id:
        raise InvalidTokenError()
    return str(user_id)
```

2. Define `TokenVerifier` protocol with a single method:
   ```python
   class TokenVerifier(Protocol):
       def verify(self, token: str) -> str:
           """Decode token and return user_id. Raise AuthError subtype on failure."""
           ...
   ```

3. Move `_decode_jwt_payload` into `UnsafeBase64Verifier` as a private static/class method, or keep it as a module-level helper. `UnsafeBase64Verifier.verify(token: str) -> str` must:
   - Call the decode logic
   - Check `exp` claim against `time.time()` → raise `ExpiredTokenError()` if expired
   - Check `user_id` claim → raise `InvalidTokenError()` if absent
   - Wrap decode errors in `InvalidTokenError() from None`
   - Return `str(user_id)` on success

   This is the exact same logic currently in `get_user_id`, extracted into the class.

4. Update `get_user_id` to:
   ```python
   async def get_user_id(request: Request, authorization: str | None = Header(None)) -> str:
       if not authorization or not authorization.startswith("Bearer "):
           raise MissingTokenError()
       token = authorization.split(" ", 1)[1].strip()
       if not token:
           raise MissingTokenError()
       verifier: TokenVerifier = request.app.state.verifier
       return verifier.verify(token)
   ```

Do NOT inline `_decode_jwt_payload`'s logic directly in `get_user_id` — that logic belongs in `UnsafeBase64Verifier.verify`. Keep `_decode_jwt_payload` as a module-level helper called by `UnsafeBase64Verifier.verify` (less churn, same result).
