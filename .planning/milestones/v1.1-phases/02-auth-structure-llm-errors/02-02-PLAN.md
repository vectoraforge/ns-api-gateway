```python
async def _invoke(self, chain, params: dict):
    for attempt in range(1, self.retry_max_attempts + 1):
        await self.circuit_breaker.before_call()
        try:
            async def operation():
                return await asyncio.wait_for(chain.ainvoke(params), timeout=self.timeout_seconds)

            response = await self.gate.run(operation)
            await self.circuit_breaker.record_success()
            return response
        except QueueFullError:
            raise
        except CircuitOpenError:
            raise
        except Exception as e:
            await self.circuit_breaker.record_failure()
            if attempt >= self.retry_max_attempts or not _is_transient_error(e):
                raise AnalysisError(str(e)) from e   # <-- THIS LINE changes
            backoff = min(
                self.retry_backoff_max_seconds,
                self.retry_backoff_base_seconds * (2 ** (attempt - 1)),
            )
            if backoff > 0:
                await asyncio.sleep(backoff)
    raise AnalysisError("LLM request failed")         # <-- AND THIS LINE changes
```

```python
def _is_transient_error(exc: Exception) -> bool:
    if isinstance(exc, (asyncio.TimeoutError, TimeoutError)):
        return True
    if isinstance(exc, (APIConnectionError, APITimeoutError, RateLimitError, InternalServerError)):
        return True
    if isinstance(exc, APIStatusError):
        status = _extract_status_code(exc)
        if status in {408, 409, 429, 500, 502, 503, 504}:
            return True
    status = _extract_status_code(exc)
    if status in {408, 409, 429, 500, 502, 503, 504}:
        return True
    return False
```

```python
class TransientLLMError(AnalysisError):
    """Raised when all retry attempts failed due to a transient LLM error.
    __cause__ holds the original exception from the last failed attempt."""
    pass


class PermanentLLMError(AnalysisError):
    """Raised when the LLM call failed with a non-transient error (no retry possible).
    __cause__ holds the original exception."""
    pass
```

Place them immediately after the `AnalysisError` class definition. They inherit `AnalysisError.__init__` which takes a plain string — no custom `__init__` needed.
